Predict with confidence interval

predict_ci <- function(model, data, var, ci = 0.95, length = 1000) {
  p <- (1 - ci) / 2
  link_inv <- model$family$linkinv
  n <- nrow(data) - length(model$coefficients)
  vals <- seq(from = min(data[[var]], na.rm = T), to = max(data[[var]], na.rm = T), length.out = length)
  df <- data.frame(col = vals)
  colnames(df) <- var
  
  pred <- data.frame(predict(model, df, se.fit = TRUE))
  pred[[var]] <- df[[var]]
  pred$lwr <- link_inv(pred$fit + qt(p, n) * pred$se.fit)
  pred$upr <- link_inv(pred$fit + qt(1 - p, n) * pred$se.fit)
  pred$fit <- link_inv(pred$fit)
  pred
}

# Weeks with no admissions must appear as zero, or a stretch of quiet weeks
# reads as continuous transmission
weekly_counts <- function(data, from = NULL, to = NULL, week_col = week) {
  obs <- data |>
    dplyr::count({{ week_col }}, name = "cases") |>
    dplyr::rename(week = {{ week_col }})
 
  from <- if (is.null(from)) min(obs$week) else as.Date(from)
  to   <- if (is.null(to))   max(obs$week) else as.Date(to)
 
  tibble(week = seq(from, to, by = "week")) |>
    left_join(obs, by = "week") |>
    mutate(cases = replace_na(cases, 0))
}
 
# One pass over the whole record. A period starts at the first of `min_run`
# consecutive weeks at or above `threshold`; two such runs separated by fewer
# than `max_gap` weeks below threshold are treated as one period.
find_outbreaks <- function(data, threshold = ob_threshold,
                           min_run = ob_min_run, max_gap = ob_max_gap,
                           week_col = week) {
  wc <- weekly_counts(data, week_col = {{ week_col }})
  r  <- rle(wc$cases >= threshold)
 
  end_i   <- cumsum(r$lengths)
  start_i <- end_i - r$lengths + 1
  seed    <- which(r$values & r$lengths >= min_run)
 
  if (length(seed) == 0) {
    return(tibble(episode = integer(), start = as.Date(character()),
                  end = as.Date(character()), weeks = integer(),
                  cases = integer(), peak_week = as.Date(character()),
                  peak_cases = integer()))
  }
 
  s <- start_i[seed]
  e <- end_i[seed]
 
  merged_s <- s[1]
  merged_e <- e[1]
  out_s <- integer()
  out_e <- integer()
 
  for (i in seq_along(s)[-1]) {
    if (s[i] - merged_e - 1 < max_gap) {
      merged_e <- e[i]
    } else {
      out_s <- c(out_s, merged_s)
      out_e <- c(out_e, merged_e)
      merged_s <- s[i]
      merged_e <- e[i]
    }
  }
  out_s <- c(out_s, merged_s)
  out_e <- c(out_e, merged_e)
 
  tibble(
    episode = seq_along(out_s),
    i_start = out_s,
    i_end   = out_e
  ) |>
    mutate(
      start = wc$week[i_start],
      end   = wc$week[i_end],
      weeks = i_end - i_start + 1L,
      cases = map2_int(i_start, i_end, \(a, b) sum(wc$cases[a:b])),
      peak_week  = map2_vec(i_start, i_end,
                            \(a, b) wc$week[a:b][which.max(wc$cases[a:b])]),
      peak_cases = map2_int(i_start, i_end, \(a, b) max(wc$cases[a:b]))
    ) |>
    select(-i_start, -i_end)
}
 
# Sensitivity of the periods to the threshold. Report this alongside the rule.
outbreak_sensitivity <- function(data, thresholds = c(3, 5, 10, 20, 30)) {
  map_dfr(thresholds, \(x) find_outbreaks(data, threshold = x) |>
            mutate(threshold = x, .before = 1))
}

# 5. Attack rate helper
#
# rate = cases on day t / (population - cumulative cases up to day t-1).
# This is a daily hazard among those not yet recorded as cases, not a
# cumulative attack rate; name it accordingly in the caption.
# ---------------------------------------------------------------------------

add_attack_rate <- function(d, group, n_days = ma_days) {
  d |>
    arrange({{ group }}, date) |>
    mutate(
      cum_cases = cumsum(n),
      lag_cum   = lag(cum_cases, default = 0),
      at_risk   = n_census - lag_cum,
      rate      = n / at_risk,
      rate_ma   = slide_dbl(rate, mean, .before = n_days - 1, .complete = TRUE),
      .by = {{ group }}
    )
}

# ---- Helpers --------------------------------------------------------
prev_correct <- function(est_val, sens, spec) {
  (est_val + spec - 1) / (sens + spec - 1)
}

clamp01 <- function(x) pmin(pmax(x, 0), 1)

# Vectorised prop.test CI (Wilson score with continuity correction)
prop_ci <- function(x, n, conf = 0.95) {
  z     <- qnorm((1 + conf) / 2)
  est   <- x / n
  yates <- pmin(0.5, abs(x - n / 2))
  z22n  <- z^2 / (2 * n)
  pc_u  <- est + yates / n
  pc_l  <- est - yates / n
  su    <- pmin(pc_u, 1)
  sl    <- pmax(pc_l, 0)
  tibble(
    ci_lwr = if_else(pc_l <= 0, 0,
                     clamp01((sl + z22n - z * sqrt(sl * (1 - sl) / n + z22n / (2 * n))) / (1 + 2 * z22n))),
    ci_upr = if_else(pc_u >= 1, 1,
                     clamp01((su + z22n + z * sqrt(su * (1 - su) / n + z22n / (2 * n))) / (1 + 2 * z22n)))
  )
}

predict_ci <- function(model, data, var, ci = 0.95, n_grid = 200) {
  link_inv <- model$family$linkinv
  df_res   <- nrow(data) - length(model$coefficients)
  tq       <- qt(1 - (1 - ci) / 2, df_res)
  rng      <- range(data[[var]], na.rm = TRUE)
  newd     <- tibble(!!var := seq(rng[1], rng[2], length.out = n_grid))
  pr       <- predict(model, newdata = newd, se.fit = TRUE, type = "link")
  newd |>
    mutate(
      fit = link_inv(pr$fit),
      lwr = link_inv(pr$fit - tq * pr$se.fit),
      upr = link_inv(pr$fit + tq * pr$se.fit)
    )
}

adjust_prev <- function(d) {
  d |>
    mutate(across(c(fit, lwr, upr),
                  \(x) clamp01(prev_correct(x, sens, spec)),
                  .names = "{.col}_true"))
}

# ===========================================================================
# 3. COUNTING ENGINE
# ===========================================================================
# Each child contributes intervals to numerator and denominator, so counting
# is a +1/-1 event table plus a cumsum: O(n_children), not O(n x weeks).
# For EVC the table gains a dose-week dimension, because
#   sum_i VE_i(w) = mult * exp(-q w) * sum_s N[w, s] exp(q s),
# so the registry is scanned once and the M draws cost two small matrix
# products per term rather than M passes over the children.
 
cover_counts_g <- function(a, b, g, ng) {
  a  <- pmax(a, 1L)
  b  <- pmin(b, n_week)
  ok <- !is.na(a) & !is.na(b) & !is.na(g) & b >= a
  a <- a[ok]; b <- b[ok]; g <- g[ok]
 
  nb  <- ng * (n_week + 1L)
  off <- (g - 1L) * (n_week + 1L)
  d   <- tabulate(off + a, nb) - tabulate(off + b + 1L, nb)
 
  m <- matrix(d, nrow = n_week + 1L, ncol = ng) |> apply(2, cumsum)
  m[seq_len(n_week), , drop = FALSE]
}
 
key_cover_counts_g <- function(a, b, g, key, ng) {
  a  <- pmax(a, 1L)
  b  <- pmin(b, n_week)
  ok <- !is.na(a) & !is.na(b) & !is.na(g) & !is.na(key) & b >= a
  a <- a[ok]; b <- b[ok]; g <- g[ok]; key <- key[ok]
 
  z   <- min(key):max(key)
  nk  <- length(z)
  k   <- key - z[1L] + 1L
  nb  <- ng * nk * (n_week + 1L)
  off <- ((g - 1L) * nk + (k - 1L)) * (n_week + 1L)
  d   <- tabulate(off + a, nb) - tabulate(off + b + 1L, nb)
 
  m <- matrix(d, nrow = n_week + 1L) |> apply(2, cumsum)
  list(n = m[seq_len(n_week), , drop = FALSE], z = z, nk = nk)
}
 
term_matrices_g <- function(a, b, g, key, mult, ng) {
  kc <- key_cover_counts_g(a, b, g, key, ng)
  Ez <- exp(outer(kc$z, q_wk))                 # n_key  x M
  Ew <- exp(outer(seq_len(n_week), -q_wk))     # n_week x M
  lapply(seq_len(ng), \(gg) {
    cols <- ((gg - 1L) * kc$nk + 1L):(gg * kc$nk)
    (kc$n[, cols, drop = FALSE] %*% Ez) * Ew * rep(mult, each = n_week)
  })
}

build_series <- function(grp) {
  lev <- levels(droplevels(grp))
  ng  <- length(lev)
  gi  <- as.integer(factor(as.character(grp), levels = lev))
 
  mk <- function(entry, exit, vax, dose_lab) {
    b   <- wk_ge(exit) - 1L
    a_d <- wk_ge(entry)
    a_n <- pmax(a_d, wk_ge(vax))      # doses before entry count from entry
    tibble(
      week    = rep(weeks, times = ng),
      group   = rep(lev, each = n_week),
      dose    = dose_lab,
      n_denom = as.vector(cover_counts_g(a_d, b, gi, ng)),
      n_vax   = as.vector(cover_counts_g(a_n, b, gi, ng))
    )
  }
 
  cov_df <- bind_rows(
    mk(child$t9,  child$t60, child$d1, "Dose 1"),
    mk(child$t18, child$t60, child$d2, "Dose 2")
  ) |>
    mutate(group    = factor(group, levels = lev),
           coverage = if_else(n_denom > 0, n_vax / n_denom, NA_real_))
 
  numM <- replicate(ng, matrix(0, n_week, m_draws), simplify = FALSE)
  for (bb in 1:2) {                                   # single-dose periods
    sel <- which(!is.na(s_first) & band_first == bb)
    if (length(sel)) numM <- Map(`+`, numM, term_matrices_g(
      pmax(a_den, s_first)[sel], b_one[sel], gi[sel], s_first[sel],
      if (bb == 1L) draws$ve1_L else draws$ve1_H, ng))
  }
  sel <- which(!is.na(s_second))                      # two-dose periods
  if (length(sel)) numM <- Map(`+`, numM, term_matrices_g(
    pmax(a_den, s_second)[sel], b_exit[sel], gi[sel], s_second[sel],
    draws$ve2, ng))
 
  denE <- cover_counts_g(a_den, b_exit, gi, ng)
 
  evc_mat <- map(seq_len(ng), \(gg) {
    dv <- denE[, gg]; dv[dv == 0] <- NA_real_
    numM[[gg]] / dv
  })
 
  evc_df <- map2(evc_mat, lev, \(E, gg) {
    qs <- apply(E, 1, quantile, probs = c(0.5, 0.025, 0.975), na.rm = TRUE)
    tibble(week = weeks, group = gg,
           evc = qs[1, ], evc_lo = qs[2, ], evc_hi = qs[3, ])
  }) |>
    bind_rows() |>
    mutate(group = factor(group, levels = lev))
 
  # ---- period summaries over the comparison window ----
  idx <- which(weeks >= win_start & weeks <= win_end)
 
  evc_win <- map2(seq_len(ng), lev, \(gg, nm) {
    dsum <- sum(denE[idx, gg])
    pa   <- if (dsum > 0) colSums(numM[[gg]][idx, , drop = FALSE]) / dsum
            else rep(NA_real_, m_draws)
    wk <- evc_df |> filter(group == nm, week >= win_start, week <= win_end)
    tibble(group = nm,
           x     = median(pa),
           xu_lo = quantile(pa, 0.025, na.rm = TRUE),
           xu_hi = quantile(pa, 0.975, na.rm = TRUE),
           x_lo  = min(wk$evc, na.rm = TRUE),
           x_hi  = max(wk$evc, na.rm = TRUE))
  }) |>
    bind_rows()
 
  cov_win <- cov_df |>
    filter(week >= win_start, week <= win_end, n_denom > 0) |>
    summarise(x = sum(n_vax) / sum(n_denom),
              x_lo = min(coverage), x_hi = max(coverage),
              .by = c(group, dose)) |>
    mutate(xu_lo = NA_real_, xu_hi = NA_real_)
 
  list(cov = cov_df, evc = evc_df, cov_win = cov_win, evc_win = evc_win)
}

make_x_breaks <- function(rng, by = "3 months", min_gap_days = 70) {
  inner <- seq(floor_date(rng[1], "month"), rng[2], by = by)
  keep  <- as.numeric(inner - rng[1]) >= min_gap_days &
           as.numeric(rng[2] - inner)  >= min_gap_days
  sort(unique(c(rng[1], inner[keep], rng[2])))
}

theil_sen <- function(x, y) {
  ok <- is.finite(x) & is.finite(y)
  x <- x[ok]; y <- y[ok]
  if (length(x) < 3) return(NULL)
  cb   <- combn(length(x), 2)
  dx   <- x[cb[2, ]] - x[cb[1, ]]

  tibble(a = median(y - b * x), b = b,
