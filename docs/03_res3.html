## 95% highest density interval (HDI): the narrowest interval that holds 95% of
## the draws, so every value inside is more probable than every value outside.
## From coda::HPDinterval() (coda is installed with BayesianTools). Returns
## numbers: convert dates back with as.Date().
hdi <- function(x, prob = 0.95) {
  h <- coda::HPDinterval(coda::as.mcmc(as.numeric(x)), prob = prob)
  c(lo = h[1, "lower"], hi = h[1, "upper"])
}

## the HDI on the log scale, for positive quantities (parameters, rates and
## ratios). The HDI depends on the scale: on the natural scale, values close to 0
## take up almost no width, so the HDI of a quantity with a long tail towards 0
## jumps between windows and looks like spikes on a log axis.
hdi_log <- function(x, prob = 0.95) exp(hdi(log(x), prob))

measles_ode <- function(t, y, p) {
  S_C <- y[["S_C"]]
  E_C <- y[["E_C"]]
  R_C <- y[["R_C"]]
  I_C_pre <- y[["I_C_pre"]]
  I_C_post <- y[["I_C_post"]]
  S_H <- y[["S_H"]]
  E_H <- y[["E_H"]]
  R_H <- y[["R_H"]]
  Q <- y[["Q"]]
  I_H_pre <- y[["I_H_pre"]]
  I_H_post <- y[["I_H_post"]]
  
  beta_C   <- p[["beta_C"]]
  beta_H  <- p[["beta_H"]]
  xi_alpha <- p[["xi_alpha"]]
  alpha   <- p[["alpha"]]
  p_a      <- p[["p_a"]]
  p_i     <- p[["p_i"]]
  sigma    <- p[["sigma"]]
  eta     <- p[["eta"]]
  gamma <- p[["gamma"]]
  delta    <- p[["delta"]]
  delta_M <- p[["delta_M"]]
  
  ## ---- forces of infection
  N_C     <- S_C + E_C + I_C_pre + I_C_post + R_C
  N_H_mix <- S_H + E_H + I_H_pre + I_H_post + R_H
  
  lambda_C <- beta_C * (I_C_pre + I_C_post) / N_C
  lambda_H <- if (N_H_mix > 0) beta_H * (I_H_pre + I_H_post) / N_H_mix else 0   # no hospital: no ward infections
  
  ## ---- community
  dS_C      <- -lambda_C * S_C - xi_alpha * S_C + delta * S_H
  
  dE_C      <-  lambda_C * S_C - sigma * E_C - xi_alpha * E_C + delta * E_H
  
  ## I_C_pre leaves by four routes: rash onset splits three ways, plus general
  ## admission. The three onset coefficients sum to eta.
  dI_C_pre  <-  sigma * E_C -
    (1 - p_a) * eta * I_C_pre -          # -> I_C_post
    p_a * (1 - p_i) * eta * I_C_pre -    # -> I_H_post
    p_a * p_i * eta * I_C_pre -          # -> Q
    xi_alpha * I_C_pre +                 # -> I_H_pre
    delta * I_H_pre                      # <- I_H_pre
  
  dI_C_post <-  (1 - p_a) * eta * I_C_pre -          # <- I_C_pre
    gamma * I_C_post
  
  dR_C      <-  gamma * I_C_post - alpha * R_C + delta * R_H + delta_M * Q
  
  ## ---- hospital
  dS_H      <-  xi_alpha * S_C - lambda_H * S_H - delta * S_H
  
  dE_H      <-  lambda_H * S_H + xi_alpha * E_C - sigma * E_H - delta * E_H
  
  ## I_H_pre leaves by three routes: rash onset splits two ways, plus discharge.
  ## The two onset coefficients sum to eta.
  dI_H_pre  <-  sigma * E_H +
    xi_alpha * I_C_pre -                 # <- I_C_pre
    (1 - p_i) * eta * I_H_pre -          # -> I_H_post
    p_i * eta * I_H_pre -                # -> Q
    delta * I_H_pre                      # -> I_C_pre
  
  dI_H_post <-  (1 - p_i) * eta * I_H_pre +          # <- I_H_pre
    p_a * (1 - p_i) * eta * I_C_pre -    # <- I_C_pre
    gamma * I_H_post
  
  dQ        <-  p_i * eta * I_H_pre +                # <- I_H_pre
    p_a * p_i * eta * I_C_pre -          # <- I_C_pre
    delta_M * Q
  
  dR_H      <-  gamma * I_H_post + alpha * R_C - delta * R_H
  
  ## ---- cumulative trackers -------------------------------------------------
  dcum_onset_C  <- eta * I_C_pre          # all community rash onsets
  dcum_onset_H  <- eta * I_H_pre          # admissions, rash appeared AFTER arrival
  dcum_noso     <- lambda_H * S_H         # infections acquired as an inpatient
  dcum_adm_rash <- p_a * eta * I_C_pre    # admissions, rash present on arrival
  dcum_inf_C    <- lambda_C * S_C         # infections acquired in the community
  
  ## names are documentation: deSolve reads this by position, so the order must
  ## match state_names. The check below enforces that.
  list(
    c(
      S_C = dS_C,
      E_C = dE_C,
      I_C_pre = dI_C_pre,
      I_C_post = dI_C_post,
      R_C = dR_C,
      S_H = dS_H,
      E_H = dE_H,
      I_H_pre = dI_H_pre,
      I_H_post = dI_H_post,
      Q = dQ,
      R_H = dR_H,
      cum_onset_C = dcum_onset_C,
      cum_onset_H = dcum_onset_H,
      cum_noso = dcum_noso,
      cum_adm_rash = dcum_adm_rash,
      cum_inf_C = dcum_inf_C
    )
  )
}

Helpers.

p_i is fixed rather than estimated: it only changes how long I_H_post transmits, which the case series barely sees. It is varied in a sensitivity analysis instead.

p_i_fixed <- 0.9

## s_C0 and s_H0 are fixed inputs, set in the Parameters section. seed is the
## number of latent infections in the community at t = 0.
init_state <- function(seed) {
  y <- c(
    S_C = s_C0 * N_C0 - seed, E_C = seed,
    I_C_pre = 0, I_C_post = 0, R_C = (1 - s_C0) * N_C0,
    S_H = s_H0 * N_H0, E_H = 0, I_H_pre = 0, I_H_post = 0, Q = 0,
    R_H = (1 - s_H0) * N_H0,
    cum_onset_C = 0, cum_onset_H = 0, cum_noso = 0,
    cum_adm_rash = 0, cum_inf_C = 0
  )
  y[state_names]
}

## ODE parameters from the fitted parameters. xi_alpha and alpha depend only on
## the fixed s_C0 and s_H0, so they are computed once, in the Parameters section.
make_params <- function(par) {
  c(beta_C = par[["beta_C"]], beta_H = par[["kappa"]] * par[["beta_C"]],
    xi_alpha = xi_alpha, alpha = alpha,
    p_a = par[["p_a"]],
    p_i = if ("p_i" %in% names(par)) par[["p_i"]] else p_i_fixed,
    delta = delta, delta_M = delta_M,
    sigma = sigma, eta = eta, gamma = gamma)
}

solve_model <- function(par, times) {
  out <- tryCatch(
    ode(y = init_state(par[["seed"]]),
        times = times, func = measles_ode,
        parms = make_params(par),
        method = "lsoda", atol = 1e-3, rtol = 1e-6),
    error = function(e) NULL
  )
  if (is.null(out) || nrow(out) != length(times) || anyNA(out)) return(NULL)
  out
}

## weekly quantities. The two case types partition all rash onsets:
##   adm_w  = p_a * onset_C + onset_H   children admitted with measles
##   out_w  = (1 - p_a) * onset_C       community cases not admitted
run_model <- function(par, n_weeks, full = FALSE) {

  sol <- solve_model(par, seq.int(0L, n_weeks * 7L, by = 7L))
  if (is.null(sol)) return(NULL)

  onset_C  <- diff(sol[, "cum_onset_C"])
  onset_H  <- diff(sol[, "cum_onset_H"])
  adm_rash <- diff(sol[, "cum_adm_rash"])

  if (!full) {
    return(list(adm = adm_rash + onset_H,
                out = onset_C - adm_rash))
  }

  keep <- -1L   # drop t = 0, so each row pairs with one diff()

  tibble(
    week     = seq_len(n_weeks),
    onset_C  = onset_C,
    onset_H  = onset_H,
    adm      = adm_rash + onset_H,
    out      = onset_C - adm_rash,
    noso     = diff(sol[, "cum_noso"]),
    inf_C    = diff(sol[, "cum_inf_C"]),
    I_H_pre  = sol[keep, "I_H_pre"],
    I_H_post = sol[keep, "I_H_post"],
    Q        = sol[keep, "Q"],
    N_H      = rowSums(sol[keep, c("S_H", "E_H", "I_H_pre",
                                   "I_H_post", "Q", "R_H")])
  )
}

Disease-free equilibrium check

With transmission switched off, hospital occupancy and hospital susceptibility must stay flat. Drift means the inputs are inconsistent and every fit inherits an artificial transient.

disease_free <- function(days = 365) {

  p <- make_params(c(beta_C = 0, kappa = 0, p_a = 0.1))   # no transmission

  ode(y = init_state(0), times = 0:days,
      func = measles_ode, parms = p,
      method = "lsoda", atol = 1e-3, rtol = 1e-6) |>

Likelihood

Weekly reported cases follow a negative binomial distribution, with mean rho times the model’s weekly rash onsets. The options above decide how the model meets the data:

  • obs_model = "total": one negative binomial for all weekly cases, with mean rho * (adm + out) and dispersion phi.
  • obs_model = "split": one negative binomial for inpatients, with mean rho * adm, and one for outpatients, with mean rho * out. Each has its own dispersion, phi_in and phi_out.
  • fit_rho = FALSE: rho is fixed at 1, so every case is assumed reported.

  m <- sapply(priors[par_names], \(p) p$r(n))

## derived quantities (thesis Section 4.3.5); s_C0, s_H0 and xi_alpha are fixed.
## One function, so the posterior draws and the MAP use the same formulas.
add_derived <- function(d) {
  d |>
    mutate(
      beta_H  = kappa * beta_C,
      D_C     = (eta + delta) / (eta + xi_alpha + delta) * (1 / eta + (1 - p_a) / gamma),
      R0C     = beta_C * D_C,             # community, everyone susceptible
      R_C_eff = s_C0 * R0C,               # community, at the start of the outbreak
      R0H     = beta_H * ((eta + xi_alpha) / (eta * (eta + xi_alpha + delta)) +
                  (1 - p_i_fixed) / gamma * (eta + xi_alpha + p_a * delta) /
                  (eta + xi_alpha + delta)),
      per_case_ratio = kappa * s_H0 / s_C0   # infections per case, H relative to C
    )
}

nb_draws <- function(mu_in, mu_out, n = 1e4) {
  if (obs_model == "total") {
    rnbinom(n, mu = mu_in + mu_out, size = p_map[["phi"]])
  } else {
    rnbinom(n, mu = mu_in,  size = p_map[["phi_in"]]) +
      rnbinom(n, mu = mu_out, size = p_map[["phi_out"]])
  }
}

## model day t is the date t0_date + 7 + t: wk2date() labels model week w with
## t0_date + 7 * w, the Monday that starts it, and week 1 starts on day 0
trajectory <- function(par, days = n_weeks * 7) {
  sol <- solve_model(par, 0:days)
  if (is.null(sol)) return(NULL)
  beta_C <- par[["beta_C"]]
  beta_H <- par[["kappa"]] * beta_C
  as.data.frame(sol) |>
    as_tibble() |>
    transmute(
      date     = t0_date + 7 + time,
      prev_C   = (I_C_pre + I_C_post) / (S_C + E_C + I_C_pre + I_C_post + R_C),
      prev_H   = (I_H_pre + I_H_post) / (S_H + E_H + I_H_pre + I_H_post + R_H),
      s_C      = S_C / (S_C + E_C + I_C_pre + I_C_post + R_C),
      s_H      = S_H / (S_H + E_H + I_H_pre + I_H_post + R_H),
      lambda_C = beta_C * prev_C,
      lambda_H = beta_H * prev_H
    )
}

R0H

The simulation set everyone in the population susceptible, and introduce 1 infectious in the pre-rash state in the community (to check R0C), or hospital (to check R0H).

## ---- closed forms from the next-generation matrix ----------------------------
r0h_formula <- function(beta_H,
                        eta,
                        gamma,
                        delta,
                        xi_alpha,
                        p_a,
                        p_i) {
  k <- eta + xi_alpha + delta
  beta_H * ((eta + xi_alpha) / (eta * k) +
              (1 - p_i) / gamma * (eta + xi_alpha + p_a * delta) / k)
}

r0c_formula <- function(beta_C, eta, gamma, delta, xi_alpha, p_a) {
  k <- eta + xi_alpha + delta
  beta_C * (eta + delta) / k * (1 / eta + (1 - p_a) / gamma)
}

## ---- the same quantities from the model ODE -----------------------------------
## seed_in: compartment of the index child ("I_H_pre" or "I_C_pre")
## count:   which infections to count ("cum_noso" or "cum_inf_C")
## Everyone is susceptible, so S_C = N_C and S_H = N_H at the start.
simulate_new_infections <- function(beta_C,
                                    beta_H,
                                    eta,
                                    gamma,
                                    delta,
                                    xi_alpha,
                                    p_a,
                                    p_i,
                                    N_C,
                                    N_H,
                                    seed_in,
                                    count) {
  y0 <- c(
    S_C = N_C,
    E_C = 0,
    I_C_pre = 0,
    I_C_post = 0,
    R_C = 0,
    S_H = N_H,
    E_H = 0,
    I_H_pre = 0,
    I_H_post = 0,
    Q = 0,
    R_H = 0,
    cum_onset_C = 0,
    cum_onset_H = 0,
    cum_noso = 0,
    cum_adm_rash = 0,
    cum_inf_C = 0
  )[state_names]
  y0[seed_in] <- 1   # one infectious child
  
  p <- c(
    beta_C = beta_C,
    beta_H = beta_H,
    xi_alpha = xi_alpha,
    alpha = 0,
    p_a = p_a,
    p_i = p_i,
    sigma = 0,
    eta = eta,
    gamma = gamma,
    delta = delta,
    delta_M = 1 / 7
  )
  
  ## long enough for the index child to finish its infectious period
  t_end <- 50 / min(eta, gamma)
  
  out <- ode(y0,
             c(0, t_end),
             measles_ode,
             p,
             rtol = 1e-10,
             atol = 1e-10)
  out[2, count]
}

## ---- R_0H and its partial derivatives ------------------------------------------
r0h <- function(beta_H, eta, gamma, delta, xi_alpha, p_a, p_i) {
  beta_H * ((eta + xi_alpha) / (eta * (eta + xi_alpha + delta)) +
              (1 - p_i) / gamma * (eta + xi_alpha + p_a * delta) / (eta + xi_alpha + delta))
}

r0h_slope <- function(par, beta_H, eta, gamma, delta, xi_alpha, p_a, p_i) {
  switch(par,
    beta_H = (eta + xi_alpha) / (eta * (eta + xi_alpha + delta)) +
               (1 - p_i) / gamma * (eta + xi_alpha + p_a * delta) / (eta + xi_alpha + delta),
    delta  = -beta_H * (eta + xi_alpha) / (eta + xi_alpha + delta)^2 *
               (1 / eta + (1 - p_i) * (1 - p_a) / gamma),
    p_a    = beta_H * (1 - p_i) * delta / (gamma * (eta + xi_alpha + delta)),
    p_i    = -beta_H * (eta + xi_alpha + p_a * delta) / (gamma * (eta + xi_alpha + delta))
  )
}

## hospital susceptibility for a given community susceptibility and xi, and xi
## for a given s_C0 and s_H0 (the inverse)
s_H_from <- function(s_C0, xi) xi * s_C0 / (1 - s_C0 + xi * s_C0)
xi_from  <- function(s_C0, s_H0) s_H0 * (1 - s_C0) / (s_C0 * (1 - s_H0))

## admission rates of immune (alpha) and susceptible (xi * alpha) children:
##   alpha = alpha_bar / (1 + s_C0 * (xi - 1))
admission <- function(s_C0, xi) {
  alpha <- alpha_bar / (1 + s_C0 * (xi - 1))
  c(alpha = alpha, xi_alpha = xi * alpha)
}

## fade-out: fewer than 0.1 latent or infectious children. Isolated children
## (Q) do not transmit, so they are not counted.
gone <- function(t, y, p) {
  sum(y[c("E_C", "I_C_pre", "I_C_post", "E_H", "I_H_pre", "I_H_post")]) - 0.1
}

## infections caused by one imported case (total), and the day of fade-out
## (t_fade: NA when the threshold is not reached within max_days, 50 years).
## counters = TRUE returns the final counters instead: infections acquired in
## the community (cum_inf_C) and in hospital (cum_noso), rash onsets and
## admissions with rash.
infections <- function(s_C0, xi, kappa, p_i, los, hospital = TRUE,
                       max_days = 365 * 50, counters = FALSE) {
  delta <- 1 / los
  N_H   <- alpha_bar * N / (delta + alpha_bar)   # inpatients that balance admissions and discharges
  s_H0  <- s_H_from(s_C0, xi)
  adm   <- admission(s_C0, xi)
  
  N_h <- if (hospital) N_H else 0
  N_c <- N - N_h
  s_c <- if (hospital) s_C0 else (s_C0 * (N - N_H) + s_H0 * N_H) / N   # inpatients join the community
  
  y0 <- c(S_C = s_c * N_c - 1, E_C = 1, I_C_pre = 0, I_C_post = 0,
          R_C = (1 - s_c) * N_c,
          S_H = s_H0 * N_h, E_H = 0, I_H_pre = 0, I_H_post = 0, Q = 0,
          R_H = (1 - s_H0) * N_h,
          cum_onset_C = 0, cum_onset_H = 0, cum_noso = 0, cum_adm_rash = 0,
          cum_inf_C = 0)[state_names]
  
  p <- c(beta_C   = beta_C,
         beta_H   = if (hospital) kappa * beta_C else 0,
         xi_alpha = if (hospital) adm[["xi_alpha"]] else 0,
         alpha    = if (hospital) adm[["alpha"]] else 0,
         p_a      = if (hospital) p_a else 0,
         delta    = if (hospital) delta else 0,
         p_i = p_i, delta_M = delta_M,
         sigma = sigma, eta = eta, gamma = gamma)
  
  out <- ode(y0, c(0, max_days), measles_ode, p, rootfunc = gone,
             atol = 1e-6, rtol = 1e-8)
  end <- out[nrow(out), ]
  if (counters) return(end[c("cum_inf_C", "cum_noso", "cum_onset_C", "cum_onset_H", "cum_adm_rash")])
  troot <- attr(out, "troot")   # the time of the root; NULL if none was found
  c(total  = end[["cum_inf_C"]] + end[["cum_noso"]],
    t_fade = if (length(troot) > 0) troot[[1]] else NA)
}

## the lowest isolation at which the hospital stops amplifying an imported case:
## it no longer adds infections (when there is an outbreak of 10 or more without
## the hospital) or no longer causes such an outbreak (when there is not).
## 0: no isolation needed; NA: amplifies even with p_i = 1.
iso_needed <- function(s_C0, s_H0, kappa) {
  xi_h   <- xi_from(s_C0, s_H0)
  none   <- infections(s_C0, xi_h, 0, 0, los_fix, hospital = FALSE)[["total"]]
  target <- if (none >= outbreak_size) none else outbreak_size
  f <- \(p) infections(s_C0, xi_h, kappa, p, los_fix)[["total"]] - target
  if (f(0) <= 0) return(0)
  if (f(1) > 0)  return(NA)
  uniroot(f, c(0, 1), tol = 0.001)$root
}
