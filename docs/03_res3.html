state_names <- c(
  "S_C", "E_C", "I_C_pre", "I_C_post", "R_C",
  "S_H", "E_H", "I_H_pre", "I_H_post", "Q", "R_H",
  "cum_onset_C", "cum_onset_H", "cum_noso", "cum_adm_rash", "cum_inf_C"
)

measles_ode <- function(t, y, p) {
  S_C <- y[["S_C"]]
  E_C <- y[["E_C"]]
  R_C <- y[["R_C"]]
  I_C_pre <- y[["I_C_pre"]]
  I_C_post <- y[["I_C_post"]]
  S_H <- y[["S_H"]]
  E_H <- y[["E_H"]]
  R_H <- y[["R_H"]]
  Q <- y[["Q"]]
  I_H_pre <- y[["I_H_pre"]]
  I_H_post <- y[["I_H_post"]]
  
  beta_C   <- p[["beta_C"]]
  beta_H  <- p[["beta_H"]]
  xi_alpha <- p[["xi_alpha"]]
  alpha   <- p[["alpha"]]
  p_a      <- p[["p_a"]]
  p_i     <- p[["p_i"]]
  sigma    <- p[["sigma"]]
  eta     <- p[["eta"]]
  gamma <- p[["gamma"]]
  delta    <- p[["delta"]]
  delta_M <- p[["delta_M"]]
  
  ## ---- forces of infection
  N_C     <- S_C + E_C + I_C_pre + I_C_post + R_C
  N_H_mix <- S_H + E_H + I_H_pre + I_H_post + R_H
  
  lambda_C <- beta_C * (I_C_pre + I_C_post) / N_C
  lambda_H <- if (N_H_mix > 0) beta_H * (I_H_pre + I_H_post) / N_H_mix else 0   # no hospital: no ward infections
  
  ## ---- community
  dS_C      <- -lambda_C * S_C - xi_alpha * S_C + delta * S_H
  
  dE_C      <-  lambda_C * S_C - sigma * E_C - xi_alpha * E_C + delta * E_H
  
  ## I_C_pre leaves by four routes: rash onset splits three ways, plus general
  ## admission. The three onset coefficients sum to eta.
  dI_C_pre  <-  sigma * E_C -
    (1 - p_a) * eta * I_C_pre -          # -> I_C_post
    p_a * (1 - p_i) * eta * I_C_pre -    # -> I_H_post
    p_a * p_i * eta * I_C_pre -          # -> Q
    xi_alpha * I_C_pre +                 # -> I_H_pre
    delta * I_H_pre                      # <- I_H_pre
  
  dI_C_post <-  (1 - p_a) * eta * I_C_pre -          # <- I_C_pre
    gamma * I_C_post
  
  dR_C      <-  gamma * I_C_post - alpha * R_C + delta * R_H + delta_M * Q
  
  ## ---- hospital
  dS_H      <-  xi_alpha * S_C - lambda_H * S_H - delta * S_H
  
  dE_H      <-  lambda_H * S_H + xi_alpha * E_C - sigma * E_H - delta * E_H
  
  ## I_H_pre leaves by three routes: rash onset splits two ways, plus discharge.
  ## The two onset coefficients sum to eta.
  dI_H_pre  <-  sigma * E_H +
    xi_alpha * I_C_pre -                 # <- I_C_pre
    (1 - p_i) * eta * I_H_pre -          # -> I_H_post
    p_i * eta * I_H_pre -                # -> Q
    delta * I_H_pre                      # -> I_C_pre
  
  dI_H_post <-  (1 - p_i) * eta * I_H_pre +          # <- I_H_pre
    p_a * (1 - p_i) * eta * I_C_pre -    # <- I_C_pre
    gamma * I_H_post
  
  dQ        <-  p_i * eta * I_H_pre +                # <- I_H_pre
    p_a * p_i * eta * I_C_pre -          # <- I_C_pre
    delta_M * Q
  
  dR_H      <-  gamma * I_H_post + alpha * R_C - delta * R_H
  
  ## ---- cumulative trackers -------------------------------------------------
  dcum_onset_C  <- eta * I_C_pre          # all community rash onsets
  dcum_onset_H  <- eta * I_H_pre          # admissions, rash appeared AFTER arrival
  dcum_noso     <- lambda_H * S_H         # infections acquired as an inpatient
  dcum_adm_rash <- p_a * eta * I_C_pre    # admissions, rash present on arrival
  dcum_inf_C    <- lambda_C * S_C         # infections acquired in the community
  
  ## names are documentation: deSolve reads this by position, so the order must
  ## match state_names. The check below enforces that.
  list(
    c(
      S_C = dS_C,
      E_C = dE_C,
      I_C_pre = dI_C_pre,
      I_C_post = dI_C_post,
      R_C = dR_C,
      S_H = dS_H,
      E_H = dE_H,
      I_H_pre = dI_H_pre,
      I_H_post = dI_H_post,
      Q = dQ,
      R_H = dR_H,
      cum_onset_C = dcum_onset_C,
      cum_onset_H = dcum_onset_H,
      cum_noso = dcum_noso,
      cum_adm_rash = dcum_adm_rash,
      cum_inf_C = dcum_inf_C
    )
  )
}

Helpers.

p_i is fixed rather than estimated: it only changes how long I_H_post transmits, which the case series barely sees. It is varied in a sensitivity analysis instead.

p_i_fixed <- 0.9

init_state <- function(s_C0, s_H0, seed) {
  y <- c(
    S_C = s_C0 * N_C0 - seed, E_C = seed,
    I_C_pre = 0, I_C_post = 0, R_C = (1 - s_C0) * N_C0,
    S_H = s_H0 * N_H0, E_H = 0, I_H_pre = 0, I_H_post = 0, Q = 0,
    R_H = (1 - s_H0) * N_H0,
    cum_onset_C = 0, cum_onset_H = 0, cum_noso = 0,
    cum_adm_rash = 0, cum_inf_C = 0
  )
  y[state_names]
}

## quantities implied by the sampled parameters
##   xi     odds ratio of susceptibility between hospital and community. This is
##          the value that makes s_C0 and s_H0 a fixed point of the disease-free
##          system, so the simulation starts without a transient.
##   alpha  admission rate for immune children, set so the population-weighted
##          rate equals the observed general admission rate alpha_bar.
##   D_inf  mean time a community case spends infectious. Cases admitted at rash
##          onset skip I_C_post, so this depends on p_a.
derived <- function(par) {
  s_C0 <- par[["s_C0"]]
  s_H0 <- par[["s_H0"]]
  
  xi       <- (s_H0 * (1 - s_C0)) / (s_C0 * (1 - s_H0))
  alpha    <- alpha_bar / (1 + s_C0 * (xi - 1))
  xi_alpha <- xi * alpha
  
  a_C    <- (sigma + delta) / (sigma + xi_alpha + delta)
  D_inf  <- (1 + eta * (1 - par[["p_a"]]) / gamma) *
    (delta + eta * a_C) / (eta * (eta + xi_alpha + delta))
  beta_C <- par[["R_eff"]] / (D_inf * s_C0)
  
  list(
    xi = xi,
    alpha = alpha,
    xi_alpha = xi_alpha,
    D_inf = D_inf,
    beta_C = beta_C,
    beta_H = par[["kappa"]] * beta_C
  )
}

make_params <- function(par, d) {
  c(beta_C = d$beta_C, beta_H = d$beta_H,
    xi_alpha = d$xi_alpha, alpha = d$alpha,
    p_a = par[["p_a"]],
    p_i = if ("p_i" %in% names(par)) par[["p_i"]] else p_i_fixed,
    delta = delta, delta_M = delta_M,
    sigma = sigma, eta = eta, gamma = gamma)
}

solve_model <- function(par, times) {
  out <- tryCatch(
    ode(y = init_state(par[["s_C0"]], par[["s_H0"]], par[["seed"]]),
        times = times, func = measles_ode,
        parms = make_params(par, derived(par)),
        method = "lsoda", atol = 1e-3, rtol = 1e-6),
    error = function(e) NULL
  )
  if (is.null(out) || nrow(out) != length(times) || anyNA(out)) return(NULL)
  out
}

## weekly quantities. The two case types partition all rash onsets:
##   adm_w  = p_a * onset_C + onset_H   children admitted with measles
##   out_w  = (1 - p_a) * onset_C       community cases not admitted
run_model <- function(par, n_weeks, full = FALSE) {

  sol <- solve_model(par, seq.int(0L, n_weeks * 7L, by = 7L))
  if (is.null(sol)) return(NULL)

  onset_C  <- diff(sol[, "cum_onset_C"])
  onset_H  <- diff(sol[, "cum_onset_H"])
  adm_rash <- diff(sol[, "cum_adm_rash"])

  if (!full) {
    return(list(adm = adm_rash + onset_H,
                out = onset_C - adm_rash))
  }

  keep <- -1L   # drop t = 0, so each row pairs with one diff()

  tibble(
    week     = seq_len(n_weeks),
    onset_C  = onset_C,
    onset_H  = onset_H,
    adm      = adm_rash + onset_H,
    out      = onset_C - adm_rash,
    noso     = diff(sol[, "cum_noso"]),
    inf_C    = diff(sol[, "cum_inf_C"]),
    I_H_pre  = sol[keep, "I_H_pre"],
    I_H_post = sol[keep, "I_H_post"],
    Q        = sol[keep, "Q"],
    N_H      = rowSums(sol[keep, c("S_H", "E_H", "I_H_pre",
                                   "I_H_post", "Q", "R_H")])
  )
}

R0H and R0C validation

The simulation set everyone in the population susceptible, and introduce 1 infectious in the pre-rash state in the community (to check R0C), or hospital (to check R0H).

## ---- closed forms from the next-generation matrix ----------------------------
r0h_formula <- function(beta_H,
                        eta,
                        gamma,
                        delta,
                        xi_alpha,
                        p_a,
                        p_i) {
  k <- eta + xi_alpha + delta
  beta_H * ((eta + xi_alpha) / (eta * k) +
              (1 - p_i) / gamma * (eta + xi_alpha + p_a * delta) / k)
}

r0c_formula <- function(beta_C, eta, gamma, delta, xi_alpha, p_a) {
  k <- eta + xi_alpha + delta
  beta_C * (eta + delta) / k * (1 / eta + (1 - p_a) / gamma)
}

## ---- the same quantities from the model ODE -----------------------------------
## seed_in: compartment of the index child ("I_H_pre" or "I_C_pre")
## count:   which infections to count ("cum_noso" or "cum_inf_C")
## Everyone is susceptible, so S_C = N_C and S_H = N_H at the start.
simulate_new_infections <- function(beta_C,
                                    beta_H,
                                    eta,
                                    gamma,
                                    delta,
                                    xi_alpha,
                                    p_a,
                                    p_i,
                                    N_C,
                                    N_H,
                                    seed_in,
                                    count) {
  y0 <- c(
    S_C = N_C,
    E_C = 0,
    I_C_pre = 0,
    I_C_post = 0,
    R_C = 0,
    S_H = N_H,
    E_H = 0,
    I_H_pre = 0,
    I_H_post = 0,
    Q = 0,
    R_H = 0,
    cum_onset_C = 0,
    cum_onset_H = 0,
    cum_noso = 0,
    cum_adm_rash = 0,
    cum_inf_C = 0
  )[state_names]
  y0[seed_in] <- 1   # one infectious child
  
  p <- c(
    beta_C = beta_C,
    beta_H = beta_H,
    xi_alpha = xi_alpha,
    alpha = 0,
    p_a = p_a,
    p_i = p_i,
    sigma = 0,
    eta = eta,
    gamma = gamma,
    delta = delta,
    delta_M = 1 / 7
  )
  
  ## long enough for the index child to finish its infectious period
  t_end <- 50 / min(eta, gamma)
  
  out <- ode(y0,
             c(0, t_end),
             measles_ode,
             p,
             rtol = 1e-10,
             atol = 1e-10)
  out[2, count]
}

Likelihood

One reporting fraction rho applies to every case, admitted or not. Two terms:

  • weekly total reported cases
  • the inpatient share of all reported cases, as a binomial on the totals

Because the same rho scales both parts of the share, it cancels from q, so p_a is pinned by the observed share and rho by the level of the weekly series. Those are separate jobs, which keeps the two well identified.

One rho is also the right choice given that the September 2024 to March 2025 vaccination campaign is not in the model. A missing campaign inflates the number of infections the model needs, and rho is a free multiplicative scale that absorbs it. Because q is a ratio, p_a is protected: the campaign scales numerator and denominator alike. With two reporting fractions and rho_in fixed, the bias would land on p_a instead.

The cost of one rho is that admissions are taken to be as under-reported as everything else, so p_a comes out near the observed inpatient share.

Quick fit by optimisation

Minutes rather than an hour. Use it to check the structure, try variants, and give the sampler a sensible starting point. The objective is the log posterior, so this is the posterior mode rather than the maximum likelihood: s_C0 and s_H0 are prior-dominated and dropping their priors would leave them unidentified.

Optimisation runs on an unconstrained scale, log for positive parameters and logit for probabilities, so optim cannot step outside the feasible region.

Two Nelder-Mead passes and no BFGS. The restart matters because Nelder-Mead collapses its simplex and stalls, and restarting from its own answer rebuilds it. BFGS is omitted deliberately: optim computes its gradients by finite differences with steps far below the solver noise from atol = 1e-3, so those gradients are mostly numerical noise.

## deSolve stops the run when this function crosses zero
make_stop_condition <- function(threshold = 1) {
  function(time, y, parms, ...)
    sum(y[infected]) - threshold
}

max_days <- 10 * 365   # safety cap, in days

run_scenario <- function(beta_C,
                         kappa,
                         p_a,
                         p_i,
                         s_C0,
                         s_H0,
                         seed,
                         los,
                         adm,
                         threshold = 1,
                         ...) {
  stopifnot(seed > threshold)   # otherwise the run stops at once
  
  N_H      <- adm * los
  N_C      <- N - N_H
  xi       <- s_H0 * (1 - s_C0) / (s_C0 * (1 - s_H0))
  alpha    <- adm / N_C / (1 + s_C0 * (xi - 1))
  xi_alpha <- xi * alpha
  
  y0 <- c(
    S_C = s_C0 * N_C - seed,
    E_C = seed,
    I_C_pre = 0,
    I_C_post = 0,
    R_C = (1 - s_C0) * N_C,
    S_H = s_H0 * N_H,
    E_H = 0,
    I_H_pre = 0,
    I_H_post = 0,
    Q = 0,
    R_H = (1 - s_H0) * N_H,
    cum_onset_C = 0,
    cum_onset_H = 0,
    cum_noso = 0,
    cum_adm_rash = 0,
    cum_inf_C = 0
  )[state_names]
  
  p <- c(
    beta_C = beta_C,
    beta_H = kappa * beta_C,
    xi_alpha = xi_alpha,
    alpha = alpha,
    p_a = p_a,
    p_i = p_i,
    delta = 1 / los,
    delta_M = delta_M,
    sigma = sigma,
    eta = eta,
    gamma = gamma
  )
  
  sim <- function(p) {
    ode(
      y0,
      0:max_days,
      measles_ode,
      p,
      method = "lsoda",
      rootfunc = make_stop_condition(threshold),
      atol = 1e-3,
      rtol = 1e-6
    )
  }
  final <- function(sol, x)
    sol[nrow(sol), x]   # value at the stop time
  
  sol  <- sim(p)
  sol0 <- sim(replace(p, "beta_H", 0))
  
  noso  <- final(sol, "cum_noso")
  inf_C <- final(sol, "cum_inf_C")
  D_inf <- 1 / (eta + xi_alpha) + eta / (eta + xi_alpha) * (1 - p_a) / gamma
  
  tibble(
    t_end     = final(sol, "time"),
    faded     = max(t_end, final(sol0, "time")) < max_days,
    R_eff     = beta_C * s_C0 * D_inf,
    noso_pct  = 100 * noso / (noso + inf_C),
    peak_prev = 1000 * max(sol[, "I_H_pre"] + sol[, "I_H_post"]) / N_H,
    inf_C     = inf_C,
    attr_C    = 100 * (inf_C - final(sol0, "cum_inf_C")) / inf_C
  )
}

## rate: ward infections per day that one infectious, non-isolated child
## spends on the ward
k_hh_parts <- function(beta_H, s_H0, los, p_i, p_a) {
  delta  <- 1 / los
  p_inf  <- sigma / (sigma + delta)       # becomes infectious before discharge
  p_rash <- p_inf * eta / (eta + delta)   # rash appears before discharge
  rate   <- beta_H * s_H0
  tibble(
    pre_rash  = rate * p_inf / (eta + delta),                   # before the rash
    post_rash = rate * p_rash * (1 - p_i) / gamma,              # rash on ward, not isolated
    readmit   = rate * (1 - p_rash) * p_a * (1 - p_i) / gamma,  # back with rash, not isolated
    K_HH      = pre_rash + post_rash + readmit
  )
}

## total infections, run to fade-out, for one set of conditions
total_infections <- function(beta_C, kappa, p_a, p_i, s_C0, s_H0, seed,
                             los, adm, threshold = 1, ...) {
  stopifnot(seed > threshold)   # otherwise the run stops at once
  
  N_H   <- adm * los
  N_C   <- N - N_H
  xi    <- s_H0 * (1 - s_C0) / (s_C0 * (1 - s_H0))
  alpha <- adm / N_C / (1 + s_C0 * (xi - 1))
  
  y0 <- c(S_C = s_C0 * N_C - seed, E_C = seed, I_C_pre = 0, I_C_post = 0,
          R_C = (1 - s_C0) * N_C,
          S_H = s_H0 * N_H, E_H = 0, I_H_pre = 0, I_H_post = 0, Q = 0,
          R_H = (1 - s_H0) * N_H,
          cum_onset_C = 0, cum_onset_H = 0, cum_noso = 0,
          cum_adm_rash = 0, cum_inf_C = 0)[state_names]
  
  p <- c(beta_C = beta_C, beta_H = kappa * beta_C,
         xi_alpha = xi * alpha, alpha = alpha, p_a = p_a, p_i = p_i,
         delta = 1 / los, delta_M = delta_M,
         sigma = sigma, eta = eta, gamma = gamma)
  
  sol <- ode(y0, 0:max_days, measles_ode, p, method = "lsoda",
             rootfunc = make_stop_condition(threshold),
             atol = 1e-3, rtol = 1e-6)
  last <- function(x) unname(sol[nrow(sol), x])   # value at the stop time
  
  tibble(total = last("cum_inf_C") + last("cum_noso"),   # infections after the start
         noso  = last("cum_noso"),
         faded = last("time") < max_days)
}

}

  if (i == 1) return(kappa[1])
  kappa[i - 1] + (0 - net[i - 1]) * (kappa[i] - kappa[i - 1]) / (net[i] - net[i - 1])
}
